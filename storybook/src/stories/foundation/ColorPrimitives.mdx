export const USAGE = typeof __COLOR_USAGE__ === 'undefined' ? {} : __COLOR_USAGE__;

export const UNPAINTED = new Set(typeof __UNPAINTED_VARIABLES__ === 'undefined' ? [] : __UNPAINTED_VARIABLES__);

export const THEMES = {
  light: ['theme-light', 'classic theme-light', 'suse theme-light'],
  dark:  ['theme-dark', 'classic theme-dark', 'suse theme-dark'],
};

export const readPrimitives = () => {
  const primitives = [];

  for (const sheet of document.styleSheets) {
    let rules = [];

    try {
      rules = sheet.cssRules;
    } catch (e) {
      continue;
    }

    for (const rule of rules) {
      if (rule.selectorText !== ':root') {
        continue;
      }

      for (const name of rule.style) {
        if (name.startsWith('--color-') && !primitives.some((p) => p.name === name)) {
          primitives.push({ name, value: rule.style.getPropertyValue(name).trim() });
        }
      }
    }
  }

  return primitives;
};

export const themeVariableNames = () => {
  const names = new Set();
  const walk = (rules) => {
    for (const rule of rules) {
      if (rule.cssRules && !rule.selectorText) {
        walk(rule.cssRules);
      } else if (rule.style && /theme-(light|dark)|\.classic|\.suse/.test(rule.selectorText || '')) {
        for (const name of rule.style) {
          if (name.startsWith('--') && !name.startsWith('--color-')) {
            names.add(name);
          }
        }
      }
    }
  };

  for (const sheet of document.styleSheets) {
    try {
      walk(sheet.cssRules);
    } catch (e) {}
  }

  return [...names];
};

export const resolveThemes = (primitives) => {
  const probe = document.createElement('span');

  document.body.appendChild(probe);
  const render = (value) => {
    probe.style.color = '';
    probe.style.color = value;

    return probe.style.color ? getComputedStyle(probe).color : null;
  };
  const byColor = new Map(primitives.map((p) => [render(p.value), p.name]));
  const colorsIn = (value) => {
    if (/^\d+\s*,\s*\d+\s*,\s*\d+(\s*,\s*[\d.]+)?$/.test(value)) {
      return [`rgba(${ value })`];
    }

    return value && CSS.supports('color', value) ? [value] : value.match(/#[0-9a-f]{3,8}\b|(rgba?|hsla?)\([^)]*\)/gi) || [];
  };
  const names = themeVariableNames();
  const original = document.body.className;
  const result = { light: new Set(), dark: new Set(), unpainted: {}, background: {} };

  for (const [mode, classes] of Object.entries(THEMES)) {
    for (const classNames of classes) {
      document.body.className = classNames;
      const style = getComputedStyle(document.body);

      for (const name of names) {
        for (const color of colorsIn(style.getPropertyValue(name).trim())) {
          const primitive = byColor.get(render(color));

          if (primitive && UNPAINTED.has(name)) {
            (result.unpainted[primitive] ||= new Set()).add(name);
          } else if (primitive) {
            result[mode].add(primitive);
          }
        }
      }
      if (classNames.startsWith('theme-')) {
        result.background[mode] = style.getPropertyValue('--body-bg').trim();
      }
    }
  }
  document.body.className = original;
  probe.remove();

  return result;
};

export const describeColor = (primitive, themes) => {
  const usage = USAGE[primitive.name] || { uses: 0, files: {}, light: false, dark: false };

  return {
    ...primitive,
    uses:      usage.uses,
    files:     Object.entries(usage.files).sort((a, b) => b[1] - a[1] || a[0].localeCompare(b[0])),
    light:     usage.light || themes.light.has(primitive.name),
    dark:      usage.dark || themes.dark.has(primitive.name),
    unpainted: [...themes.unpainted[primitive.name] || []].sort(),
    computed:  computedColors.primitives[primitive.name] || [],
    alpha:     /-a\d+$/.test(primitive.name),
  };
};

export const themeLabel = ({ uses, light, dark }) => {
  if (light && dark) {
    return 'Light + dark';
  }
  if (light || dark) {
    return light ? 'Light only' : 'Dark only';
  }

  return uses ? 'Not painted' : 'Unused';
};

export const groupByFamily = (primitives) => {
  const groups = [];

  for (const p of primitives) {
    const family = p.name.match(/^--color-([a-z]+)/)[1];
    const title = ['white', 'black'].includes(family) ? 'white and black' : family;
    let group = groups.find((g) => g.title === title);

    if (!group) {
      group = { title, primitives: [] };
      groups.push(group);
    }
    group.primitives.push(p);
  }

  return groups;
};

export const copyText = async (text) => {
  try {
    await navigator.clipboard.writeText(text);
  } catch (e) {
    const area = document.createElement('textarea');

    area.value = text;
    area.setAttribute('readonly', '');
    area.style.position = 'absolute';
    area.style.left = '-9999px';
    document.body.appendChild(area);
    area.select();
    document.execCommand('copy');
    area.remove();
  }
};

export const useScreenshots = () => {
  const [index, setIndex] = useState({});

  useEffect(() => {
    fetch('color-usage/index.json')
      .then((response) => (response.ok ? response.json() : {}))
      .then(setIndex)
      .catch(() => setIndex({}));
  }, []);

  return index;
};

export const Eyedropper = ({ color, screenshot }) => {
  const [x, y] = screenshot.tip;
  const scale = screenshot.scale || 1;
  const radians = screenshot.angle * Math.PI / 180;
  const sx = x + 68 * scale * Math.cos(radians);
  const sy = y + 68 * scale * Math.sin(radians);
  const tube = 'M0 0 L6 -3 L26 -4 L26 4 L6 3 Z';
  const fill = (value) => ({ fill: value });
  const square = (half, rx, style) => <rect x={sx - half * scale} y={sy - half * scale} width={2 * half * scale} height={2 * half * scale} rx={rx * scale} style={style} />;

  return (
    <svg className="ds-eyedropper" viewBox={`0 0 ${ screenshot.width } ${ screenshot.height }`} aria-hidden="true">
      <defs>
        <pattern id="ds-eyedropper-checker" width="8" height="8" patternUnits="userSpaceOnUse">
          <rect width="8" height="8" style={fill('var(--color-white)')} />
          <rect width="4" height="4" style={fill('var(--color-gray-96)')} />
          <rect x="4" y="4" width="4" height="4" style={fill('var(--color-gray-96)')} />
        </pattern>
      </defs>
      <g transform={`translate(${ x } ${ y }) rotate(${ screenshot.angle }) scale(${ scale })`}>
        <g style={{ fill: 'var(--color-white)', stroke: 'var(--color-white)', strokeWidth: 4, strokeLinejoin: 'round' }}>
          <path d={tube} />
          <rect x="26" y="-6" width="5" height="12" rx="1.5" />
          <rect x="31" y="-7" width="19" height="14" rx="7" />
        </g>
        <path d={tube} style={fill('url(#ds-eyedropper-checker)')} />
        <path d={tube} style={{ fill: `var(${ color.name })`, stroke: 'var(--color-gray-806)', strokeWidth: 1 }} />
        <rect x="26" y="-6" width="5" height="12" rx="1.5" style={fill('var(--color-gray-466)')} />
        <rect x="31" y="-7" width="19" height="14" rx="7" style={fill('var(--color-gray-806)')} />
      </g>
      {square(15, 7, fill('var(--color-gray-806)'))}
      {square(14, 6, fill('var(--color-white)'))}
      {square(11, 4, fill('url(#ds-eyedropper-checker)'))}
      {square(11, 4, fill(`var(${ color.name })`))}
    </svg>
  );
};

export const Computations = ({ entries }) => {
  const groups = [];

  for (const e of entries) {
    const key = `${ e.expression }|${ JSON.stringify(e.inputs) }`;
    let group = groups.find((g) => g.key === key);

    if (!group) {
      group = { key, expression: e.expression, inputs: e.inputs, places: [] };
      groups.push(group);
    }
    group.places.push(`${ e.file.split('/').pop() }${ e.themes ? ` (${ e.themes.join(', ') })` : '' }`);
  }

  return (
    <ul className="ds-list">
      {groups.map((g) => (
        <li key={g.key}>
          <code>{g.expression}</code>
          {Object.entries(g.inputs).map(([input, value]) => (
            <span key={input}> where <code>{input}</code> is <code>{value}</code></span>
          ))}
          <span className="ds-muted"> in {[...new Set(g.places)].join(', ')}</span>
        </li>
      ))}
    </ul>
  );
};

export const Primitives = () => {
  const primitives = useMemo(readPrimitives, []);
  const themes = useMemo(() => resolveThemes(primitives), [primitives]);
  const colors = useMemo(() => primitives.map((p) => describeColor(p, themes)), [primitives, themes]);
  const screenshots = useScreenshots();
  const [open, setOpen] = useState(null);
  const openColor = colors.find((c) => c.name === open);

  return (
    <div>
      <p className="ds-legend">
        {[{ light: true, dark: true, uses: 1 }, { light: true, uses: 1 }, { dark: true, uses: 1 }, { uses: 1 }].map((mode) => (
          <span key={themeMode(mode)}>
            {colors.filter((c) => themeMode(c) === themeMode(mode)).length} <ThemeBadge color={mode} />
          </span>
        ))}
        <span>{colors.filter((c) => c.computed.length).length} <span className="ds-computed">Computed</span></span>
      </p>
      {groupByFamily(colors).map((group) => (
        <div key={group.title}>
          <h2 className="ds-family">{group.title} ({group.primitives.length})</h2>
          <div className="ds-swatches">
            {group.primitives.map((c) => <Swatch key={c.name} color={c} background={themes.background} onOpen={setOpen} />)}
          </div>
        </div>
      ))}
      {openColor && (
        <Details color={openColor} background={themes.background} screenshot={screenshots[openColor.name]} screenshots={Object.keys(screenshots).length > 0} onClose={() => setOpen(null)} />
      )}
    </div>
  );
};

# Color Primitives

Use these as `var(--color-...)` instead of writing colors out or computing them from other colors. Click a swatch to see where the app paints the color: the eyedropper in the screenshot sits on a pixel of it.

- `--color-<family>-<step>`: the step is (1 - OKLCH lightness) x 1000, so a higher step is darker.
- `--color-<family>-<step>-a<n>`: the same color at n% opacity, shown over a checkerboard, the light theme background and the dark theme background.
- **Uses** counts the references in shell and @rancher/components.
- **Light + dark**, **Light only** and **Dark only** say which themes paint the color, through a theme variable or a component style. A sun marks the light theme and a moon the dark theme, and a color only the dark theme paints gets a dark badge. A dashed badge means nothing paints it: only a theme variable nothing reads, or a Sass variable that never reaches the CSS, refers to it.
- **Computed** marks a color that master computed from another color (`darken()`, `color-mix()` and the like) before it became a primitive. The details say how.

They are defined in `pkg/rancher-components/src/components/design-system/colors.scss`. A color that isn't here yet gets a new primitive in that file first.
